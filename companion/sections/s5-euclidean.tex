\section{Euclidean layer: the moment and the gap}\label{sec:euclidean}

\emph{The gap is an output: simplicity of the ground state is the only spectral input.}

This layer produces, one cutoff $g\in\CH$ at a time, the moment $M$ and the gap $m_1$
that \S\ref{sec:lppl} consumes; every limit $T\to\infty$ is taken at fixed $g$, and no gap
is assumed \rec{\S4}{22}. Theorem~5.1 \Ttext{rests on A8(iv)-T} through Theorem~D.4; the
record's \S7 uses Theorem~1.1.7 across parallel strips only (Lemma~7.4, Theorem~D.3).
Fig.~\ref{fig:slab} draws the slab.

\emph{I. Slab transfer.} With $\Om_0\equiv1$ the free vacuum,
$\psi_s:=e^{-sH(g)}\Om_0$ and $c_g:=\ip{\Om_g}{\Om_0}>0$, the FKN import A2 turns
sharp-time moments of the Euclidean field $\Phi$ under the normalised slab measure
$dq_{h_{T,g}}$ into matrix elements of $e^{-tH(g)}$ \rec{Lemma~4.1}{22}. As $T\to\infty$,
$\hat\psi_T:=\psi_T/\|\psi_T\|$ converges to $\Om_g$: $\psi_s$ is $c_g\Om_g$ plus
$e^{-sH(g)}$ applied to the component of $\Om_0$ orthogonal to $\Om_g$, whose spectral
measure has no atom at $0$ by simplicity (A1).

\begin{checkpoint}[Checkpoint I: transfer at fixed $g$]
For $T>0$, $t\in[0,2T]$ and bounded nonnegative measurable $F_1,F_2$ of the time-zero
field,
\begin{equation*}
 \int F_1\bigl(\Phi(-\tfrac t2,\cdot)\bigr)F_2\bigl(\Phi(\tfrac t2,\cdot)\bigr)dq_{h_{T,g}}
 =\frac{\ip{F_1\psi_{T-t/2}}{e^{-tH(g)}F_2\psi_{T-t/2}}}{\|\psi_T\|^2},
 \tag*{\rec{(2)}{22}}
\end{equation*}
and, as $s\to\infty$, ``$\hat\psi_s\to\Om_g$ in norm, $\|\psi_s\|\downarrow c_g$'';
``\emph{No isolation of the ground-state eigenvalue is used.}'' \rec{Lemma~4.4}{23}
\end{checkpoint}

\emph{II. The moment.} For $h$ as in (M$_2$) below, Steps~1--4 of the proof of
Theorem~5.1 write $Q_V=V(h)^2$ at time $0$ as one observable of the GJS class, which
admits products of individually Wick-ordered powers at equal times \rec{A8(ii)}{14}: the
support of $h$ meets at most three columns of GJS squares, hence at most nine ordered
column pairs, and the printed norm, with constants $K_1$ and $K_2\ge1$
\rec{A8(iii)}{15}, bounds $\|Q_V\|_{\rm GJS}$ \rec{Theorem~5.1, eq.~(5)}{24}, the bound that
Theorem~D.4 uses.
Step~5 applies eq.~(2) at $t=0$ with $F_1=F_2=|V(h)|$ and Theorem~D.4 at the slab cutoff
$h_{T,g}$, which give $\|V(h)\hat\psi_T\|\le M$ for every $T$ \rec{eq.~(6)}{25}; the
graph of $V(h)$ is weakly closed and $\hat\psi_T\to\Om_g$, so the bound passes to $\Om_g$:

\begin{checkpoint}[Checkpoint II: the uniform local moment]
Let $0\le\lambda/m_0^2<\varepsilon_{\mathscr P}$. For every $g\in\CH$ and every real
measurable $h$ with $\|h\|_\infty\le1$ and $\supp h$ contained in an interval of length
$2$, $\|V(h)\Om_g\|\le M$ (M$_2$), where
\begin{equation*}
 M^2:=M_B^2=e^{15\lambda|a_0|}\,S_2(B_{3\times5})\,9^{p'}\,V_*
 +9\,C_7\,c_U\,\overline\Sigma_{3\times5}\,\mathfrak K(p')\,\lambda^2
 \Bigl(\sum_{n=0}^{2p'}|a_n|\Bigr)^2,
 \tag*{\rec{Theorem~5.1}{24}}
\end{equation*}
$\mathfrak K(p'):=K_1K_2^{8p'}(4p')!$ [\,\dots]; $M$ depends on $(\lambda,\mathscr P,m_0)$ through $C_7$, $m_1$, $K_1$, $K_2$,
$S_2(B_{3\times5})$ and $V_*$, in working coordinates (Remark~3.3), and on nothing else.
\end{checkpoint}
Here $S_2(B_{3\times5})$ is the constant of \S\ref{sec:import} for a box of $3\times5$
squares, $V_*$ the supremum of $\|V(h')\|_{L^2(d\phi_0)}^2$ over admissible $h'$, and
$\overline\Sigma_{3\times5}=\sum_{j\ge1}(8j+12)e^{-m_1j}$ the ring sum
\recpp{\S D.e.1}{93--94}.

\emph{III. Decay at fixed $g$.} Lemmas~7.1--7.3 pass the moments of a cylinder monomial
$Q$ of sharp-time fields from the slab measures to $\Om_g$ as $T\to\infty$
\recpp{Lemmas~7.1--7.3}{26--27}. Theorem~D.3, at the one cutoff $g$, bounds their square
moments uniformly in $T$:
$\sup_{T>0}\|Q\hat\psi_T\|^2\le A\|Q^2\|_{L^2(d\phi_0)}+B'\mathfrak n(Q)^2$, with
$\mathfrak n(Q)^2$ the bound \rec{(8)}{26} on $\|Q^2\|_{\rm GJS}$ and constants $A$, $B'$
that depend on $g$ \rec{Lemma~7.1}{26}. Separately, CE2, with constants $C_7$ and $m_1$
\rec{Source Fact~2.2}{15}, applied across the strip $\{|x^0|<t/2-1\}$ of width $t-2$,
bounds the connected correlation of $Q(-t/2)$ and $Q(t/2)$ by
$C_7e^{2m_1}e^{-m_1t}\|Q\|_{\rm GJS}^2$ for $t>2$ and every cutoff \rec{Lemma~7.4}{28}.
Together, at fixed $g$:

\begin{checkpoint}[Checkpoint III: semigroup decay of centered observables]
Let $\psi_Q:=Q\Om_g-\ip{\Om_g}{Q\Om_g}\Om_g$. Then
\begin{equation*}
 \ip{\psi_Q}{e^{-tH(g)}\psi_Q}\le\widehat C_Q\,e^{-m_1t}\quad(t\ge0),
 \qquad
 \widehat C_Q:=\max\bigl\{C_7e^{2m_1}\|Q\|_{\rm GJS}^2,\;e^{3m_1}\|\psi_Q\|^2\bigr\} .
 \tag*{\rec{Proposition~7.5}{28}}
\end{equation*}
\end{checkpoint}

\emph{IV. The gap.} Decay on centred vectors controls the spectrum only if, with
$\Om_g$, they span a dense subspace. They do \rec{Theorem~7.7}{29}, by exponential
moments of the sharp-time fields in the vector state of $\Om_g$, with a constant that
depends on $g$ \rec{Lemma~7.6}{29}, which use powers of every total degree as the
source's class admits \rec{A8(ii)}{14}, strip-analytic characteristic functionals,
Fourier uniqueness and martingale convergence. For every $0\le\beta<m_1$,
Checkpoint~III forces $\mathbf 1_{[0,\beta]}(H(g))$ to annihilate every $\psi_Q$, so by
density it is the projection onto $\C\Om_g$; that is, $H(g)$ has no spectrum in
$(0,m_1)$ \rec{Theorem~7.8}{30}:

\begin{checkpoint}[Checkpoint IV: the uniform gap]
Let $0\le\lambda/m_0^2<\varepsilon_{\mathscr P}$. For every $g\in\CH$,
\begin{equation*}
 \spec H(g)\subseteq\{0\}\cup[m_1,\infty),
 \tag*{\rec{(G)}{30}}
\end{equation*}
and $0$ is a simple eigenvalue.
\end{checkpoint}

No gap entered Pieces I--III; (G) is their output. The slab limits use weak compactness in
$\HH$, and each limit is identified by a closed graph, so nothing is selected.
